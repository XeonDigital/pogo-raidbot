22def build_raid_bosses_embed(raids): 

23 """Embed of cached bosses by tier. Empty tiers are omitted instead of hiding the whole list.""" 

24 mega_legendary, mega, elite, tier5, tier3, tier1 = [], [], [], [], [], [] 

25 sha_tier5, sha_tier3, sha_tier1 = [], [], [] 

26 for boss_name, raid_data in (raids or {}).items(): 

27 entries = raid_data if isinstance(raid_data, list) else [raid_data] 

28 for data in entries: 

29 _add_raid_boss_to_fields( 

30 (boss_name or "").strip().title(), 

31 data or {}, 

32 mega_legendary, mega, elite, tier5, tier3, tier1, sha_tier5, sha_tier3, sha_tier1, 

33 ) 

34 

35 embed = discord.Embed(title="Raid Bosses") 

36 for name, bosses in ( 

37 ("Mega Legendary", mega_legendary), 

38 ("Mega", mega), 

39 ("Elite", elite), 

40 ("Tier 5", tier5), 

41 ("Tier 3", tier3), 

42 ("Tier 1", tier1), 

43 ("Shadow Tier 5", sha_tier5), 

44 ("Shadow Tier 3", sha_tier3), 

45 ("Shadow Tier 1", sha_tier1), 

46 ): 

47 if bosses: 

48 embed.add_field(name=name, value="\n".join(sorted(bosses)), inline=False) 

49 if not embed.fields: 

50 embed.description = "No raid bosses cached. Try `/refresh_raids`." 

51 return embed

54def _add_raid_boss_to_fields(display, data, mega_legendary, mega, elite, tier5, tier3, tier1, sha_tier5, sha_tier3, sha_tier1): 

55 if not display: 

56 return 

57 tier = data.get("tier") or "" 

58 pokemon = data.get("pokemon") or "" 

59 if "SHADOW" in pokemon and not display.lower().startswith("shadow "): 

60 display = f"Shadow {display}" 

61 if "SHADOW" not in pokemon: 

62 match tier: 

63 case "RAID_LEVEL_1": 

64 tier1.append(display) 

65 case "RAID_LEVEL_3": 

66 tier3.append(display) 

67 case "RAID_LEVEL_5" | "RAID_LEVEL_ULTRA_BEAST": 

68 tier5.append(display) 

69 case "RAID_LEVEL_ELITE": 

70 elite.append(display) 

71 case "RAID_LEVEL_MEGA_5": 

72 mega_legendary.append(display) 

73 case "RAID_LEVEL_MEGA": 

74 mega.append(display) 

75 else: 

76 match tier: 

77 case "RAID_LEVEL_1_SHADOW" | "RAID_LEVEL_1": 

78 sha_tier1.append(display) 

79 case "RAID_LEVEL_3_SHADOW" | "RAID_LEVEL_3": 

80 sha_tier3.append(display) 

81 case "RAID_LEVEL_5_SHADOW" | "RAID_LEVEL_5": 

82 sha_tier5.append(display)
