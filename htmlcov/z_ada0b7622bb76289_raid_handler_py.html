312async def resolve_raid_boss(interaction, bot, tier, pokemon_name, weather, invite_slots): 

313 """Infer slash tier from the current rotation, or ask the host to pick. 

314 

315 Returns (tier, pokemon_name) when listing can continue, otherwise None. 

316 """ 

317 provided_tier = optional_raid_tier(tier) 

318 matches = bot.dex.matching_rotation_bosses(pokemon_name) 

319 if provided_tier: 

320 canonical = canonical_slash_tier(provided_tier) 

321 matches = [match for match in matches if canonical and match.slash_tier == canonical] 

322 if not matches: 

323 return provided_tier, pokemon_name 

324 

325 if not matches: 

326 is_valid, response, _, _ = validate_pokemon(pokemon_name, "5") 

327 if not is_valid: 

328 await interaction.followup.send(H.guild_member_dm(interaction.guild.name, response), ephemeral=True) 

329 return None 

330 embed = discord.Embed( 

331 title="Error", 

332 description=f"That pokemon ({pokemon_name}) is not currently in rotation. If you believe this is an error, please contact TheStaplergun#6920.", 

333 ) 

334 await bot.send_ignore_error(interaction.user, " ", embed=embed) 

335 return None 

336 

337 if len(matches) > 1: 

338 await prompt_for_raid_tier(interaction, bot, pokemon_name, weather, invite_slots, matches) 

339 return None 

340 

341 match = matches[0] 

342 return match.slash_tier, listing_name_for_match(pokemon_name, match)

371async def resolve_and_validate_raid(interaction, bot, tier, pokemon_name, weather, invite_slots, already_resolved): 

372 """Return (tier, pokemon_name, weather, embed) if the raid can be listed, otherwise None.""" 

373 if not weather or str(weather).strip().lower() in ("no weather condition provided",): 

374 weather = "unknown" 

375 if not already_resolved: 

376 resolved = await resolve_raid_boss(interaction, bot, tier, pokemon_name, weather, invite_slots) 

377 if resolved is None: 

378 return None 

379 tier, pokemon_name = resolved 

380 raid_is_valid, response, suggestion = validate_and_format_message(interaction, 

381 tier, 

382 pokemon_name, 

383 weather, 

384 invite_slots) 

385 if not raid_is_valid: 

386 response += "---------\n" 

387 response += "*Here's the command you entered below. Suggestions were added. Check that it is correct and try again.*\n" 

388 correction_suggestion = f"/raid {suggestion}" 

389 await interaction.followup.send(response + "\n" + correction_suggestion) 

390 print(f'[!][{interaction.guild}][{interaction.user.name}] Raid failed to post due to invalid arguments.') 

391 return None 

392 temp = response.title 

393 temp = temp.replace("-Altered", "") 

394 temp = temp.replace("-Origin","") 

395 temp = temp.replace("-", " ") 

396 if not bot.dex.is_current_raid_boss(temp): 

397 embed = discord.Embed(title="Error", description=f"That pokemon ({temp}) is not currently in rotation. If you believe this is an error, please contact TheStaplergun#6920.") 

398 await bot.send_ignore_error(interaction.user, " ", embed=embed) 

399 return None 

400 return tier, pokemon_name, weather, response
